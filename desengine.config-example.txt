# Переименуйте этот файл в desengine.config.txt перед первым локальным запуском.
# Для первого прохода обычно достаточно:
# 1. выбрать LLM в LLM_PROVIDER;
# 2. вставить ключ только для выбранного провайдера;
# 3. остальные значения оставить как есть, если нет отдельной инструкции.

# # # # # # # # # # # # # # # # # # # # #

# LLM
# Можно выбирать, какой LLM пользоваться
# Варианты: openai | deepseek | gemini | claude | zai
# Конечно, у вас должен быть ключ к этой LLM
# И ключ этот надо записать в одну из переменных ниже
LLM_PROVIDER=deepseek

# OPENAI_API_KEY создаётся в OpenAI Platform
# Модель выбирайте попроще, чтобы не тратить токены зря
OPENAI_API_KEY=
OPENAI_MODEL=gpt-4.1
OPENAI_BASE_URL=https://api.openai.com/v1

# DEEPSEEK_API_KEY создаётся в DeepSeek Platform
# Доступ без танцев с бубном + очень дёшево
DEEPSEEK_API_KEY=
DEEPSEEK_MODEL=deepseek-v4-flash
DEEPSEEK_BASE_URL=https://api.deepseek.com

# GEMINI_API_KEY создаётся в Google AI Studio
# Сейчас это единственный бесплатный вариант
GEMINI_API_KEY=
GEMINI_MODEL=gemini-2.5-flash
GEMINI_BASE_URL=https://generativelanguage.googleapis.com/v1beta

# CLAUDE_API_KEY создаётся в Claude Console
# Claude API требует явно указать лимит ответа в токенах
CLAUDE_API_KEY=
CLAUDE_MODEL=claude-haiku-4-5
CLAUDE_BASE_URL=https://api.anthropic.com/v1
CLAUDE_MAX_TOKENS=4096

# ZAI_API_KEY создаётся в Z.AI Open Platform
# Для visual-входов выбирайте модель с поддержкой изображений, например glm-5v-turbo
ZAI_API_KEY=
ZAI_MODEL=glm-5v-turbo
ZAI_BASE_URL=https://api.z.ai/api/paas/v4
